## 2. Technology Description

- **Frontend**: React@18 + TypeScript + tailwindcss@3 + vite@5
- **3D Engine**: three@0.160 + @react-three/fiber@8 + @react-three/drei@9
- **State Management**: zustand@4
- **UI Icons**: lucide-react@0.294
- **Export Utility**: html2canvas@1.4.1
- **Data Persistence**: localStorage (无需后端)
- **Mock Data**: 预置3条典型样例数据，覆盖顺利处理/待确认/旧口径补录场景

## 4. Data Model

### 4.1 Data Model Definition

```mermaid
erDiagram
    CRACK_RECORD ||--o{ OPERATION_HISTORY : has
    CRACK_RECORD ||--o{ SOURCE_INFO : has
    
    CRACK_RECORD {
        string id PK
        string code "编号，如 YP-2024-001"
        string location "部位：叶根/叶中/叶尖"
        string position3D "3D坐标：x,y,z"
        string crackType "裂纹类型：横向/纵向/网状/疑似"
        string riskLevel "风险等级：低/中/高"
        string status "状态：pending/processing/completed/confirmed"
        string source "数据来源：gis/tablet/excel/screenshot"
        string description "裂纹描述"
        string suggestion "处理建议"
        string remark "备注"
        boolean isOldCaliber "是否旧口径"
        string createdAt "创建时间"
        string updatedAt "更新时间"
    }
    
    SOURCE_INFO {
        string id PK
        string recordId FK
        string sourceType "来源类型"
        string sourceRef "来源行号/截图标识"
        string originalData "原始数据快照"
    }
    
    OPERATION_HISTORY {
        string id PK
        string recordId FK
        string operator "操作人"
        string action "操作类型：create/update/status_change/remark"
        string detail "操作详情"
        string timestamp "操作时间"
        string cameraState "当时视角状态"
    }
    
    SCHEME {
        string id PK
        string name "方案名称"
        string description "方案描述"
        json records "标注记录快照"
        string cameraState "视角状态"
        string createdAt "创建时间"
        string updatedAt "更新时间"
    }
```

### 4.2 Type Definitions

```typescript
interface Position3D {
  x: number;
  y: number;
  z: number;
}

interface CameraState {
  position: Position3D;
  target: Position3D;
  fov: number;
}

interface SourceInfo {
  id: string;
  sourceType: 'gis' | 'tablet' | 'excel' | 'screenshot';
  sourceRef: string;
  originalData: string;
}

interface OperationHistory {
  id: string;
  operator: string;
  action: 'create' | 'update' | 'status_change' | 'remark' | 'import';
  detail: string;
  timestamp: string;
  cameraState?: CameraState;
  diff?: {
    field: string;
    oldValue: string;
    newValue: string;
  }[];
}

interface CrackRecord {
  id: string;
  code: string;
  location: 'root' | 'middle' | 'tip';
  position3D: Position3D;
  crackType: 'transverse' | 'longitudinal' | 'mesh' | 'suspected';
  riskLevel: 'low' | 'medium' | 'high';
  status: 'pending' | 'processing' | 'completed' | 'confirmed';
  source: 'gis' | 'tablet' | 'excel' | 'screenshot';
  description: string;
  suggestion: string;
  remark: string;
  isOldCaliber: boolean;
  sourceInfo: SourceInfo;
  history: OperationHistory[];
  createdAt: string;
  updatedAt: string;
}

interface Scheme {
  id: string;
  name: string;
  description: string;
  records: CrackRecord[];
  cameraState: CameraState;
  createdAt: string;
  updatedAt: string;
}

interface AppState {
  records: CrackRecord[];
  selectedRecordId: string | null;
  cameraState: CameraState;
  schemes: Scheme[];
  activeSchemeId: string | null;
  showCompleted: boolean;
  showDiffPanel: boolean;
  diffRecords: { old: CrackRecord; new: CrackRecord }[];
}
```
