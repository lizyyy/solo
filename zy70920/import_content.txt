const express = require("express");
const router = express.Router();
const upload = require("../middleware/upload");
const importService = require("../services/importService");

router.post("/samples", upload.single("file"), async (req, res) => {
  try {
    if (!req.file) return res.status(400).json({ success: false, error: "No file" });
    const { handler, batchId } = req.body;
    if (!handler || !batchId) return res.status(400).json({ success: false, error: "Missing params" });
    const result = await importService.importSampleCSV(req.file.path, { handler, batchId: parseInt(batchId) });
    res.json({ success: true, data: result });
  } catch (e) {
    res.status(400).json({ success: false, error: e.message });
  }
});
